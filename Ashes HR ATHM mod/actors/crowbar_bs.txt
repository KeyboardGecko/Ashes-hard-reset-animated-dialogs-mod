// actors/crowbar_bs.txt

actor BackstabberCan     : Inventory { Inventory.MaxAmount 1 +INVENTORY.UNDROPPABLE }
actor BackstabberFlash   : Inventory { Inventory.MaxAmount 1 +INVENTORY.UNDROPPABLE } // если решишь использовать визу-подсказку
actor BackstabberRequest : Inventory { Inventory.MaxAmount 1 +INVENTORY.UNDROPPABLE }

ACTOR Crowbar_BS : Crowbar replaces Crowbar
{
  Tag "Crowbar [BS]"

  States
  {
  Ready:
  ReadyLoop:
    TNT1 A 0 A_JumpIfInventory("lightlit", 1, "ReadyLantern_BS")
    TNT1 A 0 A_JumpIfInventory("BackstabberCan", 1, "PreHoldAlt")
    CROW A 1 A_WeaponReady(WRF_ALLOWUSER1|WRF_ALLOWUSER2|WRF_ALLOWUSER3)
    Goto ReadyLoop

  ReadyLantern_BS:
  ReadyLanternLoop:
    TNT1 A 0 A_JumpIfInventory("BackstabberCan", 1, "PreHoldAltLantern")
    SOLR A 1 BRIGHT A_WeaponReady(WRF_ALLOWUSER3|WRF_ALLOWUSER2|WRF_ALLOWUSER1)
    Goto ReadyLanternLoop

  PreHoldAlt:
    CROR A 2 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    CROR B 2 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    CROR C 2 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    Goto HoldAlt

  HoldAlt:
    TNT1 A 0 A_JumpIfInventory("BackstabberCan", 1, "HoldAltCont")
    Goto UnholdAlt
  HoldAltCont:
    CROR C 2 A_WeaponReady
    Goto HoldAlt
  UnholdAlt:
    CROR B 2 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    CROR A 2 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    Goto Ready

  PreHoldAltLantern:
    SOLR F 2 BRIGHT A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    SOLR G 2 BRIGHT A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    SOLR H 2 BRIGHT A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    Goto HoldAltLantern
  HoldAltLantern:
    TNT1 A 0 A_JumpIfInventory("BackstabberCan", 1, "HoldAltLanternCont")
    Goto UnholdAltLantern
  HoldAltLanternCont:
    SOLR H 2 BRIGHT A_WeaponReady
    Goto HoldAltLantern
  UnholdAltLantern:
    SOLR G 2 BRIGHT A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    SOLR F 2 BRIGHT A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    Goto ReadyLantern_BS

  Fire:
    TNT1 A 0 A_JumpIfInventory("lightlit", 1, "FireLantern_BS")
    TNT1 A 0 A_JumpIfInventory("BackstabberCan", 1, "FinishBackstabAlt")
    CROW NOPQ 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    Goto Downswing_BS

  FireLantern_BS:
    TNT1 A 0 A_JumpIfInventory("BackstabberCan", 1, "FinishBackstabAlt")
    SOLR FGH 3 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    Goto Downswing_BS

  Downswing_BS:
    CROW RS 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    CROW S 0 A_SetReflective
    CROW TU 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    CROW A 0 A_CustomPunch(random(3, 6) * 5, TRUE, 0, "CrowbarPuff", 90)
    CROW VW 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    CROW S 0 A_UnSetReflective
    CROW XY 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    TNT1 A 2
    TNT1 A 5 A_ReFire("Upswing_BS")
    CROR CBA 2 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    TNT1 A 0 A_JumpIfInventory("lightlit", 1, "Quicklantern_BS")
    Goto Ready

  Upswing_BS:
    TNT1 A 8
    CRO2 AB 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    CRO2 A 0 A_CustomPunch(random(3, 6) * 5, TRUE, 0, "CrowbarPuff", 90)
    CROW S 0 A_SetReflective
    CRO2 DEF 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    CROW S 0 A_UnSetReflective
    CRO2 GH 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    TNT1 A 5
    TNT1 A 5 A_ReFire("Downswing_BS")
    CROW QPON 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    TNT1 A 0 A_JumpIfInventory("lightlit", 1, "Quicklantern_BS")
    Goto Ready

  Quicklantern_BS:
    SOLR HGF 3 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    Goto Ready

  AltFire:
    TNT1 A 0 A_JumpIfInventory("lightlit", 1, "AltFireLantern_BS")
    TNT1 A 0 A_JumpIfInventory("BackstabberCan", 1, "FinishBackstabAlt")
    CROR ABC 2 A_Light0
    Goto HeavyAttack_BS

  AltFireLantern_BS:
    TNT1 A 0 A_JumpIfInventory("BackstabberCan", 1, "FinishBackstabAlt")
    SOLR FGH 3 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    Goto HeavyAttack_BS

  HeavyAttack_BS:
    TNT1 A 5
    CROW BBCCCD 1
    CROW E 1
    {
      A_CustomPunch(random(12,12) * 5, TRUE, 0, "CrowbarPuff", 90);
      A_FireCustomMissile("Effect_ParryTokenGiver", 0, 0, 0, 0, 0);
    }
    CROW FGHI 1
    TNT1 A 8
    TNT1 A 0 A_JumpIfInventory("lightlit",1,"Quicklantern_BS")
    CROW QPON 2
    CROW A 1
    CROW A 1 A_ReFire
    Goto Ready

  FinishBackstabAlt:
    TNT1 A 0 A_TakeInventory("BackstabberCan", 1)
    TNT1 A 0 A_GiveInventory("BackstabberRequest", 1)
    CROW E 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    CROW FGHI 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    TNT1 A 2
    CROW QPON 2 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
    Goto Ready
  }
}
