OptionValue "JMAnimScaleVals"
{
    0, "$JM_SIZE_BIG"
    1, "$JM_SIZE_NORMAL"
    2, "$JM_SIZE_SMALL"
	3, "$JM_SIZE_TINY"
}

OptionMenu "JMAnimOptions"
{
    Title "$JM_OPT_ANIM_TITLE"
	StaticText ""
    Option "$JM_OPT_ANIM_ENABLE", "jm_anim_enabled", "OnOff"
    Option "$JM_OPT_ANIM_PLAY",   "jm_anim_play",    "OnOff", "jm_anim_enabled", 0
	StaticText ""
    Option "$JM_OPT_WIN_SIZE", "jm_anim_scale", "JMAnimScaleVals", "jm_anim_enabled"
    StaticText ""
    Option "$JM_VOICE_PLAY", "jm_voice_enable", "OnOff"
    Slider "$JM_VOICE_VOL",  "jm_voice_volume", 0.0, 1.0, 0.05, 2, "jm_voice_enable"
}

AddOptionMenu "OptionsMenu"
{
    StaticText ""
    Submenu "$JM_OPT_ANIM_SUBMENU", "JMAnimOptions"
}
